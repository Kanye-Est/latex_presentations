\section{1. 威胁模型与攻击直觉 (Threat Model)}
% ==========================================

\begin{frame}{1.1 重新认识属性推断 (Property Inference)}
    \textbf{核心定义：} 模型所有者发布了一个训练好的模型，攻击者试图通过分析模型参数，反向推导出\textbf{训练集中未公开的宏观统计分布}（通常与模型主任务无关）。
    
    \vspace{0.3cm}
    \begin{tcolorbox}[colback=blue!5,colframe=myblue,title=隐私攻击的“宏观”与“微观”]
    \begin{itemize}
        \item \textbf{微观级别 (Micro-level):} 成员推断 (Membership)、模型逆向 (Model Inversion)。它们关心的是\textbf{特定的个体}（如：张三的数据是否被用了？）。
        \item \textbf{宏观级别 (Macro-level):} 属性推断 (Property/Distribution)。它关心的是\textbf{群体的统计特征}（如：训练集中女性占比是否超过 60\%？）。
    \end{itemize}
    \end{tcolorbox}
    
    \textbf{为什么危害巨大？} 
    可能泄露企业的核心商业机密（如客户群体的受众画像、数据采集环境），或用于审计模型是否使用了存在偏见/违规的数据集。
\end{frame}

\begin{frame}{1.2 论文的威胁模型设定 (White-box Assumption)}
    \textbf{攻击者的能力 (Capabilities):}
    \begin{itemize}
        \item \textbf{白盒访问 (White-box):} 攻击者拥有目标模型（Target Model）的完整架构和\textbf{所有权重参数 (Weights \& Biases)}。
        \item \textbf{辅助数据 (Auxiliary Data):} 攻击者拥有与受害者训练集分布相似的辅助数据集，用于后续构造影子模型。
    \end{itemize}

    \textbf{攻击者的目标 (Objective):}
    \begin{itemize}
        \item 判断目标模型的训练集 $D$ 是否具备某一特定属性 $P$（二分类问题），或者推断该属性的具体比例（回归问题）。
    \end{itemize}
\end{frame}

\section{2. 核心挑战：深扒排列不变性 (Permutation Invariance)}
% ==========================================

\begin{frame}{2.1 朴素的白盒攻击：展平基线 (Flatten Baseline)}
    \textbf{直觉出发点：} 在完全白盒的假设下，攻击者拥有目标模型的所有权重 (Weights) 和偏置 (Biases)。最直接、最符合直觉的攻击方式是什么？
    
    \begin{center}
        \Large{\textbf{提取全部参数 $\rightarrow$ 展平为一维向量 $\rightarrow$ 训练二分类元模型 (Meta-Classifier)}}
    \end{center}
    
    \textbf{论文中的定义：} 作者将其作为\textbf{基线模型 (Baseline)}。也就是把网络所有层的参数 $W_1, b_1, W_2, b_2 \dots$ 简单粗暴地 concatenate 成一个超长向量 $V \in \mathbb{R}^d$。
    
    \vspace{0.3cm}
    \begin{tcolorbox}[colback=red!5,colframe=myred,title=直觉的崩塌 (Why it fails?)]
        实验表明，这种朴素的 Flatten 方法在推断宏观属性时，准确率几乎等同于随机猜测（$\sim 50\%$）。\\
        \textbf{原因：} 深度学习优化的随机性（初始化、SGD 打乱），导致即使是在完全相同的分布上训练出的两个模型，其内部也会产生巨大的\textbf{“位置噪音” (Position Noise)}。
    \end{tcolorbox}
\end{frame}

\begin{frame}{2.2 直觉的崩塌：Flatten 失败的三大元凶}
    \begin{itemize}
        \setlength{\itemsep}{8pt}
        \item \textbf{1. 欧氏空间的“距离欺骗”：}
        \begin{itemize}
            \item 元分类器判断两个样本是否同类的依据是高维特征空间中的欧氏距离。
            \item 若模型 A 的关键特征由第 1 个神经元负责，模型 B 由第 100 个负责，展平后这两者的距离极其遥远，导致元分类器被表象欺骗。
        \end{itemize}
        \item \textbf{2. 恐怖的阶乘爆炸 ($n!$)：}
        \begin{itemize}
            \item 仅一个含 100 个神经元的隐藏层，就有 $100!$ 种（约 $9.3 \times 10^{157}$）等效排列。
            \item 同一“属性信号”在展平空间中会产生 $100!$ 个影子变体。仅靠几百个影子模型训练无异于盲人摸象，根本无法泛化。
        \end{itemize}
        \item \textbf{3. 信号被噪音彻底淹没：}
        \begin{itemize}
            \item “权重在哪个位置”（排列噪音）占据了巨大的数值方差，而“权重代表什么属性”（真实信号）仅引起微小的波动。
            \item \textit{犹如在震耳欲聋的重金属音乐中，试图听清一只蚊子的叫声。}
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{2.3 致命痛点：同功不同构 (Permutation Invariance)}
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{如图所示的神经元互换：}
            \begin{itemize}
                \item 假设我们交换隐藏层的两个节点位置。
                \item 其对应的入边权重和出边权重会发生交叉变换。
                \item \textbf{结果：} 尽管内部的权重矩阵大变样，但网络最终的前向预测输出 $y$ 丝毫不受影响。
            \end{itemize}
            \vspace{0.2cm}
            \textit{这也意味着如果直接展平权重，巨大的“位置噪音”会彻底淹没微弱的属性信号。}
        \end{column}
        \begin{column}{0.55\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{fig:permutation.png}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.4 数学推导 (上)：同功不同构的根源}
    为了理解“位置噪音”的破坏力，我们从前向传播的数学公式入手。
    
    对于一个全连接网络，第 $i$ 隐藏层的前向计算如下：
    \begin{equation}
        h^{(i)} = \sigma\left(W^{(i)} h^{(i-1)} + b^{(i)}\right)
    \end{equation}
    
    \vspace{0.2cm}
    \textbf{引入排列操作：} 假设该层有 $n$ 个神经元，我们引入一个 $n \times n$ 的\textbf{排列矩阵 (Permutation Matrix)} $P$。$P$ 的特性是正交阵，即 $P^T P = P P^T = I$。
    
    现在，由于训练的随机性，第 $i$ 层的神经元位置被打乱了（本质是对向量 $h^{(i)}$ 进行行交换）：
    \begin{equation}
        \tilde{h}^{(i)} = P h^{(i)} = P \cdot \sigma\left(W^{(i)} h^{(i-1)} + b^{(i)}\right) = \sigma\left( (\textcolor{myblue}{P W^{(i)}}) h^{(i-1)} + (\textcolor{myblue}{P b^{(i)}}) \right)
    \end{equation}
    
    \textbf{初步结论：} 一次神经元的互换，导致当前层的权重变成了 $\tilde{W}^{(i)} = P W^{(i)}$，偏置变成了 $\tilde{b}^{(i)} = P b^{(i)}$。由于矩阵左乘，这表现为\textbf{行交换 (Row Swapping)}。
\end{frame}

\begin{frame}{2.5 数学推导 (下)：联动补偿与参数雪崩}
    但是，网络是一个整体。第 $i$ 层神经元被打乱了，下一层 ($i+1$ 层) 接收到的输入变成了 $\tilde{h}^{(i)}$。
    为了保证网络\textbf{最终的分类输出完全不变}，第 $i+1$ 层的权重必须进行补偿：
    
    \begin{equation}
        \text{原始下一层前向：} \quad W^{(i+1)} h^{(i)}
    \end{equation}
    \begin{equation}
        \text{打乱后的下一层前向：} \quad \tilde{W}^{(i+1)} \tilde{h}^{(i)} = \tilde{W}^{(i+1)} (P h^{(i)})
    \end{equation}
    为了使两者相等：$\tilde{W}^{(i+1)} P h^{(i)} = W^{(i+1)} h^{(i)}$，等式右侧巧妙地插入 $I = P^T P$：
    \begin{equation}
        \tilde{W}^{(i+1)} P h^{(i)} = W^{(i+1)} (P^T P) h^{(i)} = (W^{(i+1)} P^T) (P h^{(i)})
    \end{equation}
    由此得出 \textcolor{myred}{$\tilde{W}^{(i+1)} = W^{(i+1)} P^T$}。由于矩阵右乘，这表现为\textbf{列交换 (Column Swapping)}。
    
    \vspace{0.2cm}
    \textbf{灾难性后果：} 一次简单的神经元位置互换 $P$，会导致当前层权重\textbf{行交换}，下一层权重\textbf{列交换}。功能完全等效的模型，在参数空间上已经“面目全非”！
\end{frame}

\begin{frame}{2.6 聚焦属性推断：困境与破局目标}
    \textbf{现在我们将数学推导映射回属性推断 (PIA) 场景：}
    
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \begin{itemize}
                \item \textbf{Meta-Classifier 的梦魇：} 
                刚才的数学推导证实了，模型权重的展平（Flatten）包含了错综复杂的行/列交换。
                \item 面对如此巨大的排列空间，原本代表微弱属性分布的特征信号，被噪音产生了不可逾越的鸿沟（距离欺骗）。
            \end{itemize}
        \end{column}
        \begin{column}{0.55\textwidth}
            \begin{tcolorbox}[colback=yellow!10,colframe=orange,title=我们需要做什么？(Our Goal)]
                我们必须设计一种映射函数（或特征表示） $\mathcal{F}(\text{weights})$，使得无论内部经历怎样的排列矩阵 $P$ 变换，其输出的特征都是唯一且恒定的：\\
                \vspace{0.2cm}
                \centerline{$\mathcal{F}(W^{(i)}, W^{(i+1)}) \equiv \mathcal{F}(PW^{(i)}, W^{(i+1)}P^T)$}
                \vspace{0.2cm}
                
                \textbf{这也就是接下来要介绍的核心技术：}\\
                \textbf{排列不变表示 (Permutation Invariant Representation)}
            \end{tcolorbox}
        \end{column}
    \end{columns}
\end{frame}

\section{3. 破局之道：设计排列不变表示}
% ==========================================

\begin{frame}{3.1 解决方案一：强制节点排序 (Canonical Form)}
    \textbf{核心思路：人为制定规则，把打乱的神经元强行“对齐”。}
    
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{算法 1 深度拆解：}
            \begin{itemize}
                \item \textbf{计算指标：} 遍历第 $t$ 层，计算节点特征 $\mathcal{M}$ (如权范数) 并得出排序索引 $\sigma$ (行 4-6)。
                \item \textbf{重排与联动补偿 (关键)：}
                \begin{itemize}
                    \item \textbf{行 7：} $h'_t \leftarrow \sigma(h_t)$，对当前层的神经元进行\textbf{行交换}。
                    \item \textbf{行 9：} $w^{t+1}_{j*} \leftarrow \sigma(w^{t+1}_{j*})$，必须对下一层的对应权重进行\textbf{列交换}，以保证网络功能等效！
                \end{itemize}
                \item \textbf{局限：} 严重依赖启发式指标，网络较深时容易产生排序抖动。
            \end{itemize}
        \end{column}
        \begin{column}{0.55\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{fig:alg1.png}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{3.2 解决方案二：引入 DeepSets 抹除顺序}
    \textbf{核心思路：既然排不好序，就干脆把网络层视为“无序集合 (Set)”。}
    
    Zaheer 等人的 DeepSets 理论证明：任何定义在集合上的排列不变函数，都可以分解为：
    \begin{equation}
        f(X) = \rho\left( \sum_{x \in X} \phi(x) \right)
    \end{equation}
    \begin{itemize}
        \item $\phi(x)$：特征提取网络，独立作用于集合中的每个元素。
        \item $\sum$：\textbf{对称聚合操作}。利用加法交换律 ($a+b=b+a$) 彻底抹除元素的顺序！
        \item $\rho(\cdot)$：读出网络，接收聚合后的总体特征，输出属性预测。
    \end{itemize}
\end{frame}

\begin{frame}{3.3 DeepSets 架构工作流 (Figure 5)}
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{化矩阵为“无序集合”：}
            \begin{enumerate}
                \item \textbf{Flattening：} 提取节点 $i$ 的入边权重 $w_i$、偏置 $b_i$ 及其上下文打包为一个独立向量。
                \item \textbf{Processing ($\phi$)：} 通过 $\phi$ 网络独立提取节点特征。
                \item \textbf{Summation ($\Sigma$)：} \textcolor{myred}{\textbf{核心魔法！}} 对同层所有节点特征进行\textbf{求和}。由于 $A+B=B+A$，强行抹除了节点的排列顺序噪音。
                \item \textbf{Prediction ($\rho$)：} 拼接各层特征 $L_i$ 送入 $\rho$ 分类器预测最终属性 $P / \overline{P}$。
            \end{enumerate}
        \end{column}
        \begin{column}{0.55\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth,height=0.8\textheight,keepaspectratio]{fig:meta-classifier.png}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{3.4 代码化理解：DeepSets 算法流程}
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{算法 2 与架构图的对应关系：}
            \begin{itemize}
                \item \textbf{特征提取 (行 5-6)：} 遍历每层的节点 $i$，将其参数喂入 $\phi_t$ 获取节点特征 $N^t$。\\(\textit{对应图 5 的 Node Processing})
                \item \textbf{对称聚合 (行 7)：} $L_t \leftarrow \sum N_i^t$，这就是图中的 Layer Summation，实现排列不变性的关键步骤。
                \item \textbf{最终预测 (行 8-9)：} 拼接所有层的集合特征 $\mathcal{F}$，通过读出网络 $\rho(\mathcal{F})$ 输出预测结果。\\(\textit{对应图 5 的 Prediction})
            \end{itemize}
