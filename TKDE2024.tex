            \vspace{0.2cm}

            \begin{itemize}
                \item \textbf{非独立同分布 (Non-IID)：} 如图所示，Party 1 和 Party N 的本地数据分布存在极大差异（分别在左侧和右侧）。
                \item \textbf{分布泄露：} 好奇的中心服务器在聚合全局模型时，通过暗中分析各方上传的特定梯度，即可反向推断出各参与方高度机密的\textbf{本地数据分布特征}。
            \end{itemize}

\section{2. 动机与挑战：从参数到梯度 (Motivation)}
% ==========================================

\begin{frame}{2.1 战场转移：从单一白盒到联邦交互}
    与针对单一白盒模型的静态权重推断不同，本篇论文面临着全新的挑战环境：
    
    \vspace{0.3cm}
    \begin{itemize}
        \setlength{\itemsep}{10pt}
        \item \textbf{动态的梯度视角：} 攻击者不再拥有最终的静态模型权重，而是只能观察到训练过程中每一轮的\textbf{动态梯度更新 ($\Delta W$)}。
        \item \textbf{无先验知识要求 (Zero Prior Knowledge)：} 传统的属性推断通常需要攻击者具备辅助数据集来训练影子模型，而本文致力于在无辅助数据的情况下直接恢复分布。
    \end{itemize}
\end{frame}

\begin{frame}{2.2 为什么差分隐私 (DP) 防不住分布泄露？}
    工业界通常采用 \textbf{DP-SGD} 来防御反演攻击，即在梯度中注入高斯噪音隐藏个体贡献。
    
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{DP 的盲区 (DP's Blind Spot)：}
            \begin{itemize}
                \item \textbf{微观保护：} DP 严格保护了\textbf{特定个体}不被识别。
                \item \textbf{宏观暴露：} DP 的理论基础在于\textbf{保留群体的统计可用性}。这意味着，代表宏观统计规律的“数据分布”，在加上 DP 噪音后，依然能被强力的推断算法重构出来！
            \end{itemize}
        \end{column}

\section{3. 核心攻击方法论 (Attack Methodology)}
% ==========================================

\begin{frame}{3.1 核心攻击方法：PGDA 工作流}
    论文提出基于优化的 \textbf{PGDA (Property/Prior Gradient Distribution Attack)} 框架来逆推数据分布。
    
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{攻击机制解析：}
            \begin{enumerate}
                \item \textbf{前向传播提取特征：} 攻击者初始化虚拟的输入数据分布 (Input data)，分别将其输入到目标客户端的局部模型 (Local model) 与全局模型 (Global model) 中，得到相应的输出表示 $z_k$ 与 $z_G$。
                \item \textbf{构建损失函数：} 计算两者的差异 $\text{Loss}(z_k, z_G)$，该损失本质上度量了虚拟数据分布产生的梯度与截获的真实梯度之间的距离。
                \item \textbf{反向传播优化数据：} 通过 Backpropagation，不断更新优化虚拟输入数据的分布参数，直至其产生的梯度与客户端真实上传的梯度一致，从而实现分布重构。
            \end{enumerate}
        \end{column}
        \begin{column}{0.55\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{fig_pgda.png}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{3.2 PGDA 核心：基于梯度匹配的分布优化}
    \begin{tcolorbox}[colback=yellow!5,colframe=orange,title=PGDA 的数学本质]
        攻击者并非直接还原数据，而是求解一个\textbf{梯度匹配 (Gradient Matching)} 问题。
    \end{tcolorbox}
    
    \textbf{1. 目标函数 (Objective Function)：}
    攻击者通过最小化以下优化目标来逼近真实分布：
    \begin{equation}
        \min_{\mathbf{D}_{dummy}} \mathcal{L}_{match} = \left\| \nabla W(\mathbf{D}_{dummy}, \theta) - \nabla W_{intercepted} \right\|^2 + \alpha \cdot \mathcal{R}(\mathbf{D}_{dummy})
    \end{equation}
    
    \textbf{2. 关键算子解析：}
    \begin{itemize}
        \item $\nabla W_{intercepted}$：中心服务器从客户端截获的真实梯度更新。
        \item $\mathbf{D}_{dummy}$：攻击者构造的虚拟数据分布。
        \item $\mathcal{R}(\cdot)$：正则化项，用于保证生成的分布符合现实约束。
    \end{itemize}
    
    \textbf{结论：} 当梯度完全匹配时，虚拟分布 $\mathbf{D}_{dummy}$ 即可代表客户端的真实宏观分布，攻击者从而窃取了隐私。
\end{frame}

\section{4. 实验评估与防御分析 (Evaluation \& Defense)}
% ==========================================

\begin{frame}{4.1 突破最强防御：DP-SGD 实验验证}
    \textbf{本论文最具冲击力的实证结果之一：}
    
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{防御失效：}
            \begin{itemize}
                \item 在客户端开启 DP-SGD 并注入不同级别的隐私噪音（不同 $\epsilon$ 值）后执行攻击。
                \item \textbf{结论：} 尽管噪音足以掩盖单一样本特征，但由于“分布”是全局统计量，对零均值噪音具有天然的鲁棒性。攻击者依然能精准测算出客户端数据分布。
            \end{itemize}
        \end{column}
        \begin{column}{0.55\textwidth}

\section{5. 总结与启示 (Conclusion)}
% ==========================================

\begin{frame}{5.1 研究启示与未来防御方向}
    \begin{tcolorbox}[colback=blue!5,colframe=myblue,title=核心结论 (Takeaways)]
        \begin{itemize}
            \item \textbf{盲点暴露：} 现有的联邦学习隐私保护（如 DP）主要聚焦于“微观个体”，严重忽视了跨孤岛场景下“宏观分布”泄露的商业危害。
            \item \textbf{梯度的双刃剑：} 梯度不仅传递了模型优化的方向，也潜藏了丰富的数据全局统计特征。
        \end{itemize}
    \end{tcolorbox}
    
    \vspace{0.3cm}
    \textbf{未来的潜在防御思路：}
    \begin{itemize}
        \item 针对梯度的分布混淆技术 (Distribution Obfuscation)。
        \item 结合安全多方计算 (SMPC) 的高级聚合策略。
    \end{itemize}
\end{frame}
