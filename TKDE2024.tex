\section{1. 前置知识：FL 与 DP 的数学基础}
% ==========================================

\begin{frame}{1.1 联邦平均算法 (FedAvg) 机制}
    联邦学习通过在不交换原始数据的情况下协作训练模型，其核心是 FedAvg 算法：
    
    \vspace{0.2cm}
    \begin{tcolorbox}[colback=blue!5,colframe=myblue,title=角色动作分解]
        \begin{itemize}
            \item \textbf{客户端（本地训练）：} 寻找最优参数 $\boldsymbol{\theta}_k^{(t)}$ 以最小化本地经验风险：
            \begin{equation}
                \boldsymbol{\theta}_k^{(t)} = \arg\min_{\boldsymbol{\theta}} \frac{1}{n_k} \sum_{i=1}^{n_k} \ell(f(\boldsymbol{x}_i; \boldsymbol{\theta}), y_i) + R(\boldsymbol{\theta})
            \end{equation}
            \item \textbf{服务器（全局聚合）：} 按照数据量占比对各方上传的参数进行加权平均：
            \begin{equation}
                \boldsymbol{\theta}_G^{(t)} \leftarrow \sum_{k=1}^N \frac{n_k}{n} \boldsymbol{\theta}_k^{(t)}
            \end{equation}
        \end{itemize}
    \end{tcolorbox}
    \textit{注：服务器截获的正是客户端每一轮上传的“赃物” $\boldsymbol{\theta}_k^{(t)}$。}
\end{frame}

\begin{frame}{1.2 差分隐私的理论局限：Group DP 与分布泄露}
    差分隐私 (DP) 虽然保护了微观个体，但在保护宏观分布时面临“数学死局”：
    
    \begin{itemize}
        \item \textbf{个体的防御 ($s=1$)：} DP 保证增加/删除一条记录对输出影响微乎其微（受 $e^\epsilon$ 限制）。
        \item \textbf{群体的泄露 ($s$ records)：} 为了改变数据分布，通常涉及 $s$ 条记录的变动。根据 \textbf{Group DP} 性质，隐私边界呈指数级爆炸：
        \begin{equation}
            \Pr[\mathcal{A}(D) \in \mathcal{O}] \le \mathbf{e^{s\epsilon}} \cdot \Pr[\mathcal{A}(D'') \in \mathcal{O}]
        \end{equation}
        \item \textbf{死局：} 若要强制掩盖 $s=1000$ 人的分布差异，必须将 $\epsilon$ 设得极小（如 $1/s$），这会导致注入的加噪量“排山倒海”，使模型精度（Utility）彻底归零。
    \end{itemize}
\end{frame}

\section{2. 问题定义与核心动机}
% ==========================================

\begin{frame}{2.1 分布推断的核心动机：捕捉“偏见”}
    \textbf{为什么能从参数中看穿分布？—— 利用“本地偏好”与“全局中庸”的分歧。}
    
    \begin{columns}
        \begin{column}{0.6\textwidth}
            \begin{itemize}
                \item \textbf{本地模型的“偏科”：} 客户端 $k$ 的模型仅在本地数据分布（如 $[-1, 3]$）上表现优异，在缺失分布上会产生极大的预测偏差。
                \item \textbf{全局模型的“全能”：} 聚合后的模型见多识广（如 $[-3, 3]$），对全域都有稳定的泛化能力。
                \item \textbf{攻击点：} 寻找本地模型与全局模型“吵架”（分歧最大）的地方，即可反推出该客户端缺失了哪些分布的数据。
            \end{itemize}
        \end{column}
        \begin{column}{0.4\textwidth}
            \begin{center}
                \includegraphics[width=\textwidth]{TKDE2024pic/fig_fl_threat.png}
            \end{center}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{2.2 攻击目标与特权服务器}
    \begin{itemize}
        \item \textbf{威胁模型：} 诚实但好奇 (Honest-but-curious) 的服务器。不破坏协议，但记录一切。
        \item \textbf{输入已知：} 
        \begin{enumerate}
            \item $\mathcal{M}_k = \{\boldsymbol{\theta}_k^{(t)}\}$：第 $k$ 方上传的所有历史参数序列。
            \item $\boldsymbol{\theta}_G$：最终的全局模型。
        \end{enumerate}
        \item \textbf{终极目标：} 在**零先验知识**（无影子数据集）的情况下，精准还原参与方 $k$ 训练数据的特征分布 $P(X)$ 和标签分布 $P(Y)$。
    \end{itemize}
\end{frame}

\begin{frame}{3.1 第一步：参数时序聚合 (Algorithm 1)}
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{难点：} 如何从 $T$ 轮历史更新中提炼最关键的特征？
            
            \vspace{0.2cm}
            \textbf{启发式权重聚合策略：}
            \begin{enumerate}
                \item \textbf{算分歧 (行 2)：} 计算每轮本地参数与全局模型的欧氏距离：$C_k^{(t)} = || \boldsymbol{\theta}_k^{(t)} - \boldsymbol{\theta}_G ||_2$。
                \item \textbf{分权重 (行 4)：} 利用 \textbf{Softmax} 转换，分歧越大的轮次赋予越高权重。
                \item \textbf{得输入 (行 5)：} 得到“浓缩版”本地参数 $\theta_k$。
            \end{enumerate}

\begin{frame}{3.2 方案 A：针对可微模型的 PGDA (Algorithm 2)}
    \begin{columns}
        \begin{column}{0.45\textwidth}
            \textbf{Projected Gradient Descent Attack：}
            \begin{itemize}
                \item \textbf{核心逻辑：} 求解一个**梯度匹配 (Gradient Matching)** 优化问题。
                \item \textbf{流程：} 随机初始化虚拟分布 $\rightarrow$ 反向传播调整数据分布 $\rightarrow$ 使虚拟梯度不断逼近截获的真实梯度。
                \item \textbf{鲁棒性：} 引入高斯扰动筛选在噪音下依然表现一致的合成样本。
            \end{itemize}

\begin{frame}{3.5 PGDA 数学本质：基于梯度匹配的优化}
    攻击者通过最小化以下损失函数来“拟合”出真实的本地分布参数 $\mathbf{D}_{dummy}$：
    \begin{equation}
        \min_{\mathbf{D}_{dummy}} \mathcal{L}_{match} = \left\| \nabla W(\mathbf{D}_{dummy}, \theta) - \nabla W_{intercepted} \right\|^2 + \alpha \cdot \mathcal{R}(\mathbf{D}_{dummy})
    \end{equation}
    
    \begin{itemize}
        \item \textbf{物理含义：} 寻找一组数据分布，使其在当前模型 $\theta$ 下产生的梯度，与被截获的梯度 $\nabla W_{intercepted}$ 尽可能重合。
        \item \textbf{正则化项 $\mathcal{R}$：} 确保生成的分布满足基本的概率分布约束。
    \end{itemize}
\end{frame}

\section{4. 实验评估与防御启示}
% ==========================================

\begin{frame}{4.1 击穿最强防御：为什么 DP 失效了？}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{实验观察：}
            \begin{itemize}
                \item \textbf{噪音的抵消效应：} 虽然 DP 给每个梯度加了高斯噪音，但由于分布推断计算的是大量样本的**均值/方差**，零均值的加性噪音在聚合运算中被自然“过滤”掉了。
                \item \textbf{结论：} 现有的差分隐私防御，面对基于“统计平均”的分布推断攻击时，显得力不从心。
            \end{itemize}
        \end{column}

\section{5. 总结与 Q\&A}

\begin{frame}{5.1 研究启示}
    \begin{tcolorbox}[colback=blue!5,colframe=myblue,title=核心结论 (Takeaways)]
        \begin{itemize}
            \item \textbf{隐私盲点：} 跨孤岛场景下，宏观商业情报（分布）的泄露危害不亚于个体隐私泄露。
            \item \textbf{算法鲁棒性：} PGDA 证明了分布信息具有极强的抗噪性，单一的 DP 机制不足以提供全面保护。
            \item \textbf{未来方向：} 需要研究针对分布级信息的混淆机制。
        \end{itemize}
    \end{tcolorbox}
